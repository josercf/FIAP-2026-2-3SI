Hands-on Lab: LogiTech Enterprise

Missão

Implementar um servidor Node.js que envia atualizações de entrega da LogiTech via SSE.

Passo a Passo

1. Criar branch `feature/sse-tracking`
2. Escrever `server.js`
3. Commit e merge.

// server.js
const http = require('http');

http.createServer((req, res) => {
  if (req.url === '/events') {
    res.writeHead(200, {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection': 'keep-alive'
    });
    setInterval(() => {
      res.write(`data: ${JSON.stringify({ status: 'Em trânsito', lat: -23.5, lng: -46.6 })}\n\n`);
    }, 2000);
  }
}).listen(3000, () => console.log('SSE Server na porta 3000'));
